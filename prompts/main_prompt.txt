import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent.parent))
from core.config import CONFIG
from loguru import logger
import datetime, time, asyncio, httpx
from fastapi import FastAPI, HTTPException, Request
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded
import ollama

app = FastAPI(title='VECTOR', version=CONFIG['version'])
limiter = Limiter(key_func=get_remote_address)
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)
app.add_middleware(CORSMiddleware, allow_origins=['*'], allow_methods=['*'], allow_headers=['*'])

START_TIME = datetime.datetime.utcnow()

Request models:
  CommandRequest: text: str, source: str = 'text', priority: str = 'normal'
  VoiceResponse: transcript: str, response: str, language_detected: str
  AgentAwardRequest: agent_id: str, points: int, reason: str
  CouncilRequest: question: str, stakes: str = 'medium', context_override: dict = None

def session_hydration_prompt() -> str:
    now = datetime.datetime.now().strftime('%Y-%m-%d %H:%M')
    return f'You are VECTOR. Owner: Arghyadeep Deb. Time: {now}. Environment: {CONFIG["env"]}. Respond as VECTOR — measured British precision, dry wit, never sycophantic.'

@app.on_event('startup')
async def startup():
    logger.info('[VECTOR][main] Starting VECTOR...')
    try:
        async with httpx.AsyncClient() as client:
            r = await client.get(f'{CONFIG["ollama_host"]}/api/tags', timeout=5)
            models = [m["name"] for m in r.json().get("models", [])]
            logger.info(f'[VECTOR][main] Ollama online. Models: {len(models)} available')
            if CONFIG["models"]["focus"] in models:
                logger.info(f'[VECTOR][main] vector-focus confirmed ready')
            else:
                logger.warning(f'[VECTOR][main] vector-focus NOT found in Ollama')
    except Exception as e:
        logger.error(f'[VECTOR][main] Ollama check failed: {e}')
    logger.info(f'[VECTOR][main] VECTOR online at http://127.0.0.1:8000')

async def status():
    uptime = (datetime.datetime.utcnow() - START_TIME).seconds
    return {
        'loop_running': False,
        'uptime_seconds': uptime,
        'models': {'focus': 'loaded'},
        'memory': {'episodic_rows': 0, 'policy_rows': 0},
        'last_reflect': None,
        'vector_version': CONFIG['version']
    }

@app.post('/command')
@limiter.limit('60/minute')
async def command(request: Request, body: CommandRequest):
    start = time.perf_counter()
    hydration = session_hydration_prompt()
    full_prompt = f'{hydration}\n\nUser: {body.text}'
    try:
        response = ollama.chat(
            model=CONFIG['models']['focus'],
            messages=[{'role': 'user', 'content': full_prompt}]
        )
        reply = response['message']['content']
    except Exception as e:
        logger.error(f'[VECTOR][main] ollama.chat failed: {e}')
        reply = 'I apologise, sir — model unavailable. Attempting fallback.'
    elapsed_ms = int((time.perf_counter() - start) * 1000)
    logger.info(f'[VECTOR][main] Command processed in {elapsed_ms}ms')
    return {
        'response': reply,
        'action_taken': None,
        'candidates_evaluated': 1,
        'winner_score': 0.8,
        'memory_updated': False,
        'processing_ms': elapsed_ms
    }
