**Approach:** Move MCP pre-import guard to `core/__init__.py` (runs at VECTOR startup), add a verification check.

**Files:**
- Modify: `core/__init__.py`

- [ ] **Step 1: Add MCP pre-import guard**

Replace `core/__init__.py` content with:

```python
"""
V.E.C.T.O.R. core package.
Core systems: config, memory, inference, loop, voice, city, sensing, control, brain.
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent.parent))
from loguru import logger

logger.info("[VECTOR] core package loaded")

# Pre-import MCP SDK before vectorcode/src goes on sys.path to prevent
# vectorcode/src/mcp/ from shadowing the real mcp package.
for _pkg in ("mcp", "fastmcp"):
    try:
        __import__(_pkg)
    except ImportError:
        pass

if "mcp" in sys.modules and not hasattr(sys.modules["mcp"], "server"):
    logger.warning("[VECTOR] mcp module may be shadowed by vectorcode/src/mcp/")

__all__ = [
    "config", "memory", "inference", "loop", "voice",
    "city", "sensing", "control", "brain",
]
```

### Task 3: #26 — Fix Subprocess Bridge 2000-char Cap

**Files:**

```python
def _run_ipc(args):
    """JSON stdin/stdout IPC mode for VECTOR bridge."""
    import json
    try:
        data = json.loads(sys.stdin.read())
    except Exception as e:
        json.dump({"status": "error", "output": f"IPC parse error: {e}", "exit_code": -1}, sys.stdout)
        sys.stdout.flush()
        return
    task = data.get("task", "").strip()
    if not task:
        json.dump({"status": "error", "output": "Empty task", "exit_code": -1}, sys.stdout)
        sys.stdout.flush()
        return

    from repl import VectorREPL, _ask_ollama, _save_session
    project_dir = Path(getattr(args, "dir", None) or ".").resolve()
    repl = VectorREPL(project_dir=project_dir)
    if getattr(args, "model", None):
        repl.session.current_model = args.model
    if getattr(args, "mode", None):
        try:
            repl.session.set_mode(args.mode)
        except ValueError as e:
            print(f"[VECTOR] {e}")

    repl._history.append({"role": "user", "content": task})
    reply = _ask_ollama(repl.session.current_model, repl._history)
    repl._history.append({"role": "assistant", "content": reply})
    _save_session(repl._history, repl.session.current_model)

    json.dump({"status": "ok", "output": reply, "exit_code": 0}, sys.stdout)
    sys.stdout.flush()
```

- [ ] **Step 4: Rewrite `run_task` in bridge to use IPC**

In `districts/coding/vectorcode_bridge.py`, replace the `run_task` method (lines 58-113):

```python
    async def run_task(self, task: str, mode: str = "run", timeout: int = 300) -> dict:
        if not self._vc_available:
            return {"status": "unavailable", "output": "VectorCode not found", "exit_code": -1, "task": task}

        if not isinstance(task, str) or not task.strip():
            return {"status": "error", "output": "Empty or invalid task", "exit_code": -1, "task": ""}
        task = task.strip()

        if mode not in ("run", "auto", "hybrid"):
            mode = "run"

        import json
        payload = json.dumps({"task": task, "mode": mode}).encode("utf-8")
        cmd = [sys.executable, str(_VC_ENTRY), "run", "--ipc"]

        logger.info(f"[VectorCode] Running: {mode} — {task[:60]}...")

        proc = await asyncio.create_subprocess_exec(
            *cmd,
            stdin=asyncio.subprocess.PIPE,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
            cwd=str(Path(__file__).parent.parent),
        )
        try:
            stdout, stderr = await asyncio.wait_for(proc.communicate(payload), timeout=timeout)
        except asyncio.TimeoutError:
            return {"status": "timeout", "output": f"Task timed out after {timeout}s", "exit_code": -1, "task": task}
        except Exception as e:
            logger.error(f"[VectorCode] Bridge error: {e}")
            return {"status": "error", "output": str(e), "exit_code": -1, "task": task}

        err = stderr.decode("utf-8", errors="replace").strip()
        try:
            result = json.loads(stdout.decode("utf-8", errors="replace").strip())
        except json.JSONDecodeError:
            return {"status": "error", "output": err or stdout.decode(), "exit_code": -1, "task": task}

        if result.get("exit_code") == 0:
            output = result.get("output", "")
            logger.info(f"[VectorCode] Task completed: {task[:40]}")
            try:
                from core.memory.episodic import EpisodicMemory
                mem = EpisodicMemory()
                mem.store(
                    event_type="vectorcode_task",
                    content=f"Task: {task[:100]}\nResult: {output[:200]}",
                    tags=["vectorcode", "coding", mode],
                    source="vectorcode_bridge",
                )
            except Exception as e:
                logger.warning(f"[VectorCode] Memory store failed: {e}")
            return {"status": "ok", "output": output, "exit_code": 0, "task": task}
        else:
            logger.warning(f"[VectorCode] Task failed: {err[:100]}")
            return {"status": "error", "output": err or result.get("output", ""), "exit_code": result.get("exit_code", -1), "task": task}
```
